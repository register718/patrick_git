% Colour marks: \mk{<colour id>}{<text>} highlights a statement of the model text in the colour of the
% passage that supports it in the cited paper (sources_marked/<module>/<paper>.pdf).
% The colour ids and RGB values are those of tools/mark_pdfs.py; the claim behind each id is listed
% per module in tools/marks/<module>.json.  Remove the \input{marks} line to typeset without marks.
\definecolor{mk1}{RGB}{255,235,59}   % yellow
\definecolor{mk2}{RGB}{139,220,110}  % green
\definecolor{mk3}{RGB}{100,215,240}  % cyan
\definecolor{mk4}{RGB}{255,150,200}  % pink
\definecolor{mk5}{RGB}{255,178,80}   % orange
\definecolor{mk6}{RGB}{200,160,255}  % violet
\definecolor{mk7}{RGB}{215,180,130}  % tan
\definecolor{mk8}{RGB}{140,170,255}  % blue
\definecolor{mk9}{RGB}{255,120,110}  % salmon
\definecolor{mk10}{RGB}{ 60,215,190} % teal
\newcommand{\mk}[2]{\begingroup\sethlcolor{mk#1}\hl{#2}\endgroup}
% \mkt: highlight for text in table cells (soul works there, but \cite/\ref must stay outside of the argument).
\newcommand{\mkt}[2]{\begingroup\sethlcolor{mk#1}\hl{#2}\endgroup}
% \mkb: highlight box that does not break lines, for short entries such as reactions and IDs.
\newcommand{\mkb}[2]{{\setlength{\fboxsep}{0.5pt}\colorbox{mk#1}{#2}}}
% Reaction tables: \mkx{c1}{c2}{text} marks the reaction in the colour of its first statement and appends a small
% square in the colour of a second one; \mkxx adds a third square.  Colours of the text/paper marks of the same module.
\newcommand{\mksq}[1]{\hspace{1pt}\colorbox{mk#1}{\rule{0pt}{1.6ex}\hspace{2.5pt}}}
\newcommand{\mkx}[3]{\mkb{#1}{#3}\mksq{#2}}
\newcommand{\mkxx}[4]{\mkb{#1}{#4}\mksq{#2}\mksq{#3}}
% \mkn{c1}{c2,c3,...}{text}: reaction (or table entry) marked in colour c1, followed by one small square per further colour.
\makeatletter
\newcommand{\mkn}[3]{\mkb{#1}{#3}\@for\mkc:=#2\do{\mksq{\mkc}}}
\makeatother
